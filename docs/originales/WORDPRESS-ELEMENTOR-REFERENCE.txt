/* ==========================================================
   MEDAL — Elementor frontend JS
   Video/audio + inquiry form + interactive mesh halos.
   ========================================================== */

(() => {
  const root = document.querySelector('.medal-site');
  if (!root || root.dataset.medalReady === '1') return;
  root.dataset.medalReady = '1';

  const CONFIG = {
    formEndpoint: 'https://formsubmit.co/ajax/marcelo@medalusa.com',
    fallbackEmail: 'marcelo@medalusa.com',
    subject: 'New inquiry from medalusa.com',
  };

  /* ---------- Video / sound ---------- */
  (() => {
    const video = root.querySelector('#hero-video');
    const toggle = root.querySelector('#sound-toggle');
    const label = root.querySelector('#sound-label');
    if (!video || !toggle || !label) return;

    const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    video.muted = true;
    video.volume = 1;

    if (!reduceMotion) {
      video.play().catch(() => {});
    } else {
      video.pause();
    }

    toggle.addEventListener('click', () => {
      video.muted = !video.muted;
      const on = !video.muted;
      toggle.setAttribute('aria-pressed', String(on));
      label.textContent = on ? 'Sound on' : 'Sound off';
      if (on) video.play().catch(() => {});
    });

    document.addEventListener('visibilitychange', () => {
      if (!document.hidden && video.paused && !reduceMotion) {
        video.play().catch(() => {});
      }
    });
  })();

  /* ---------- Inquiry form ---------- */
  (() => {
    const form = root.querySelector('#contact-form');
    const statusEl = root.querySelector('#form-status');
    const button = root.querySelector('#cf-submit');
    if (!form || !statusEl || !button) return;

    const setStatus = (text, type = '') => {
      statusEl.textContent = text;
      statusEl.className = `form-status${type ? ` is-${type}` : ''}`;
    };

    form.querySelectorAll('input, textarea').forEach((el) => {
      el.addEventListener('input', () => {
        el.closest('.field')?.classList.remove('is-invalid');
        if (statusEl.classList.contains('is-error')) setStatus('');
      });
    });

    const validate = () => {
      let firstInvalid = null;

      form.querySelectorAll('[required]').forEach((el) => {
        const ok = el.value.trim() !== '' && el.checkValidity();
        el.closest('.field')?.classList.toggle('is-invalid', !ok);
        if (!ok && !firstInvalid) firstInvalid = el;
      });

      if (!firstInvalid) return true;

      const email = form.elements.email;
      const message = firstInvalid === email && email.value.trim()
        ? 'Enter a valid email address.'
        : 'Complete your name, work email and a few lines about your brand.';

      setStatus(message, 'error');
      firstInvalid.focus();
      return false;
    };

    form.addEventListener('submit', async (event) => {
      event.preventDefault();

      if (form.elements._honey.value) return;
      if (!validate()) return;

      button.setAttribute('aria-busy', 'true');
      button.disabled = true;
      setStatus('Sending…');

      const data = {
        name: form.elements.name.value.trim(),
        email: form.elements.email.value.trim(),
        brand: form.elements.brand.value.trim(),
        message: form.elements.brand.value.trim(),
        _subject: CONFIG.subject,
        _replyto: form.elements.email.value.trim(),
        _template: 'table',
        _captcha: 'false',
      };

      try {
        const response = await fetch(CONFIG.formEndpoint, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            Accept: 'application/json',
          },
          body: JSON.stringify(data),
        });

        const json = await response.json().catch(() => ({}));
        if (!response.ok || String(json.success) === 'false') {
          throw new Error(json.message || response.statusText);
        }

        form.reset();
        setStatus('Thank you. Your inquiry was sent successfully.', 'ok');
      } catch (error) {
        console.error('[Medal form]', error);
        setStatus(`We could not send your message. Write to us directly at ${CONFIG.fallbackEmail}.`, 'error');
      } finally {
        button.removeAttribute('aria-busy');
        button.disabled = false;
      }
    });
  })();

  /* ---------- Interactive halos ---------- */
  (() => {
    const section = root.querySelector('.message');
    const form = root.querySelector('.contact-form');
    const topOrb = root.querySelector('.mesh-orb--top');
    const bottomOrb = root.querySelector('.mesh-orb--bottom');
    if (!section || !topOrb || !bottomOrb) return;

    const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    const mobileQuery = window.matchMedia('(max-width: 720px)');

    const FOLLOW_SPEED = 0.055;
    const RETURN_SPEED = 0.018;
    const CORE_MAX_RADIUS = 92;
    const CORE_MIN_RADIUS = 58;

    const pointer = {
      x: 0,
      y: 0,
      inside: false,
      overForm: false,
    };

    const orbs = [
      {
        el: topOrb,
        duration: 52000,
        startX: .72,
        startY: -.28,
        travelX: -.72,
        travelY: .12,
        phase: 0,
        x: 0,
        y: 0,
        autoX: 0,
        autoY: 0,
        initialized: false,
        insideCore: false,
        lastCoreEnter: 0,
      },
      {
        el: bottomOrb,
        duration: 58000,
        startX: -.24,
        startY: .58,
        travelX: .82,
        travelY: -.78,
        phase: .22,
        x: 0,
        y: 0,
        autoX: 0,
        autoY: 0,
        initialized: false,
        insideCore: false,
        lastCoreEnter: 0,
      },
    ];

    let activeOrb = null;

    const clamp = (value, min, max) => Math.min(Math.max(value, min), max);
    const smoothStep = (t) => t * t * (3 - 2 * t);
    const pingPong = (value) => {
      const t = value % 1;
      return t < .5 ? smoothStep(t * 2) : smoothStep((1 - t) * 2);
    };

    const getCoreRadius = (orb) => clamp(orb.el.offsetWidth * .115, CORE_MIN_RADIUS, CORE_MAX_RADIUS);

    const getOrbCenter = (orb) => ({
      x: orb.x + orb.el.offsetWidth / 2,
      y: orb.y + orb.el.offsetHeight / 2,
    });

    const distanceToOrb = (orb) => {
      const center = getOrbCenter(orb);
      return Math.hypot(pointer.x - center.x, pointer.y - center.y);
    };

    const releaseOrb = () => {
      activeOrb = null;
      orbs.forEach((orb) => { orb.insideCore = false; });
    };

    const evaluateCores = () => {
      if (mobileQuery.matches || !pointer.inside || pointer.overForm) return;

      const now = performance.now();
      const candidates = [];

      orbs.forEach((orb, index) => {
        const distance = distanceToOrb(orb);
        const radius = getCoreRadius(orb);
        const isInside = distance <= radius;

        if (isInside && !orb.insideCore) {
          orb.lastCoreEnter = now;
        }

        orb.insideCore = isInside;

        if (isInside) {
          candidates.push({
            index,
            distance,
            lastCoreEnter: orb.lastCoreEnter,
          });
        }
      });

      if (!candidates.length) return;

      if (candidates.length === 1) {
        activeOrb = candidates[0].index;
        return;
      }

      candidates.sort((a, b) => {
        if (b.lastCoreEnter !== a.lastCoreEnter) {
          return b.lastCoreEnter - a.lastCoreEnter;
        }
        return a.distance - b.distance;
      });

      activeOrb = candidates[0].index;
    };

    const updatePointer = (event) => {
      const rect = section.getBoundingClientRect();
      pointer.x = clamp(event.clientX - rect.left, 0, rect.width);
      pointer.y = clamp(event.clientY - rect.top, 0, rect.height);
    };

    section.addEventListener('pointerenter', (event) => {
      if (mobileQuery.matches) return;
      pointer.inside = true;
      updatePointer(event);
    });

    section.addEventListener('pointermove', (event) => {
      if (mobileQuery.matches) return;

      pointer.inside = true;
      pointer.overForm = event.target instanceof Element && Boolean(event.target.closest('.contact-form'));
      updatePointer(event);

      if (pointer.overForm) {
        releaseOrb();
        return;
      }

      evaluateCores();
    });

    section.addEventListener('pointerleave', () => {
      pointer.inside = false;
      pointer.overForm = false;
      releaseOrb();
    });

    if (form) {
      form.addEventListener('pointerenter', () => {
        pointer.overForm = true;
        releaseOrb();
      });

      form.addEventListener('pointerleave', () => {
        pointer.overForm = false;
      });
    }

    const animate = (time) => {
      const rect = section.getBoundingClientRect();
      const width = rect.width;
      const height = rect.height;
      const isMobile = mobileQuery.matches;
      const reduced = reduceMotion.matches;

      orbs.forEach((orb, index) => {
        const progress = reduced ? .2 : (time / orb.duration + orb.phase) % 1;
        const path = pingPong(progress);

        let autoX = width * (orb.startX + orb.travelX * path);
        let autoY = height * (orb.startY + orb.travelY * path);

        if (!reduced) {
          autoX += Math.sin(time / 9200 + index * 2.3) * 22;
          autoY += Math.cos(time / 10800 + index * 1.7) * 18;
        }

        orb.autoX = autoX;
        orb.autoY = autoY;

        let targetX = autoX;
        let targetY = autoY;

        if (!isMobile && pointer.inside && !pointer.overForm && activeOrb === index) {
          targetX = pointer.x - orb.el.offsetWidth / 2;
          targetY = pointer.y - orb.el.offsetHeight / 2;

          targetX = clamp(
            targetX,
            -orb.el.offsetWidth * .58,
            width - orb.el.offsetWidth * .42
          );

          targetY = clamp(
            targetY,
            -orb.el.offsetHeight * .58,
            height - orb.el.offsetHeight * .42
          );
        }

        if (!orb.initialized) {
          orb.x = autoX;
          orb.y = autoY;
          orb.initialized = true;
        }

        const followsPointer = activeOrb === index && pointer.inside && !pointer.overForm && !isMobile;
        const easing = followsPointer ? FOLLOW_SPEED : RETURN_SPEED;

        orb.x += (targetX - orb.x) * easing;
        orb.y += (targetY - orb.y) * easing;

        orb.el.style.transform = `translate3d(${orb.x}px, ${orb.y}px, 0)`;
      });

      requestAnimationFrame(animate);
    };

    requestAnimationFrame(animate);
  })();
})();
